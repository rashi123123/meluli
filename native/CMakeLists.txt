cmake_minimum_required(VERSION 3.20)
project(meluli_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED WHISPER_SOURCE OR NOT DEFINED JSON_INCLUDE)
  message(FATAL_ERROR "Use scripts/build-cpp-worker.mjs to provide pinned source dependencies")
endif()
# Keep developer machine paths out of assertion strings and debug metadata.
get_filename_component(MELULI_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${MELULI_SOURCE_ROOT}=/meluli-src" "-fdebug-prefix-map=${MELULI_SOURCE_ROOT}=/meluli-src")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(meluli-asr-worker whisper-worker.cpp)
target_include_directories(meluli-asr-worker PRIVATE "${JSON_INCLUDE}")
target_link_libraries(meluli-asr-worker PRIVATE whisper)
add_executable(meluli-worker-guard-test EXCLUDE_FROM_ALL tests/worker-guard-test.cpp)
target_include_directories(meluli-worker-guard-test PRIVATE "${JSON_INCLUDE}")
target_link_libraries(meluli-worker-guard-test PRIVATE whisper)
add_executable(meluli-latency-probe EXCLUDE_FROM_ALL tests/latency-probe.cpp)
target_include_directories(meluli-latency-probe PRIVATE "${JSON_INCLUDE}")
target_link_libraries(meluli-latency-probe PRIVATE whisper)
